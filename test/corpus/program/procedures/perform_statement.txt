Perform subroutine
======
PERFORM select_sflight TABLES sflight_tab 
                       USING  p_carr p_conn. 
---

(source
  (perform_statement
    (subroutine_spec
      (identifier))
    (argument_list
      (positional_argument
        (identifier)))
    (argument_list
      (positional_argument
        (identifier))
      (positional_argument
        (identifier)))))

======
Perform subroutine in program
======
PERFORM (subr) IN PROGRAM (prog) IF FOUND. 
---

(source
  (perform_statement
    routine: (subroutine_spec
      name: (dynamic_spec
        value: (identifier))

        name: (dynamic_spec
          value: (identifier)))
      (if_found))))

======

      (if_found))))
